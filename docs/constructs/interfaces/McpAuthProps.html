    allowedRedirectUris: string[]

    Allowed redirect URIs for dynamic client registration. Clients registering with URIs not in this list are rejected.

    ReadonlyapiDomain

    apiDomain: string

    The API domain that acts as the MCP OAuth issuer. All OAuth discovery endpoints will reference this domain. Example: 'api.example.com'

    ReadonlyauthorizeEndpoint

    authorizeEndpoint: string

    The upstream OAuth authorize endpoint URL. The authorize Lambda proxies (302 redirects) to this URL. Example: 'https://auth.example.com/oauth2/authorize'

    ReadonlyclientId

    clientId: string
